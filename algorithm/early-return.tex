\begin{algorithm}[H]
    \caption{Early-Exit Edges Detection via Conditional Return-Value Flow}
    \label{algo:runtime-shape-check-detection}
    \footnotesize
    \SetKwFunction{RunOnFunction}{RunOnFunction}
    \SetKwFunction{MarkEarlyExitEdge}{MarkEarlyExitEdge}
    \SetKwFunction{GetReturnInst}{GetReturnInst}
    \SetKwFunction{GetReturnValue}{GetReturnValue}
    \SetKwFunction{BuildCRVFG}{BuildCRVFG}
    \SetKwFunction{GetCondition}{GetCondition}
    \SetKwFunction{GetSourceValue}{GetSourceValue}
    \SetKwFunction{GetSourceBlock}{GetSourceBlock}
    \SetKwFunction{FindBranchWithCondition}{FindBranchWithCondition}
    \SetKwFunction{GetSuccessor}{GetSuccessor}
    \SetKwFunction{IsError}{IsError}
    \SetKwFunction{IsCall}{IsCall}
    \SetKwFunction{GetCalleesInCondition}{GetCalleesInCondition}
    \SetKwProg{Fn}{Function}{:}{}
    
    \Fn{\RunOnFunction{$F,\;\mathcal{E},\;\mathcal{S}$}}{
        $inst_{ret} \gets$ \GetReturnInst{$F$}\;
        $v_{ret} \gets$ \GetReturnValue{$inst_{ret}$}\;
        $P \gets$ \BuildCRVFG{$F, v_{ret}$}\;
        
        \ForEach{path $p \in P$}{
            $\varphi \gets$ \GetCondition{$p$}\;
            $v \gets$ \GetSourceValue{$p$}\;
            $B_{src} \gets$ \GetSourceBlock{$p$}\;
            
            \uIf{\IsError{$v, \varphi, \mathcal{E}, \mathcal{S}$}}{
                $br \gets$ \FindBranchWithCondition{$B_{src}, \varphi$}\;
                \If{$br \neq \text{null}$}{
                    $B_{dst} \gets$ \GetSuccessor{$br, \varphi$}\;
                    \MarkEarlyExitEdge{$B_{src} \rightarrow B_{dst}$}\;
                }
            }
            \uElseIf{\IsCall{$v$}}{
                \RunOnFunction{callee of $v$, $\mathcal{E},\ \mathcal{S}$}\;
            }
        }
        \tcp{Recursively analyze callees referenced in conditions}
        $C_{\varphi} \gets$ \GetCalleesInCondition{$P$}\;
        \ForEach{callee $f \in C_{\varphi}$}{
            \RunOnFunction{$f$, $\mathcal{E},\ \mathcal{S}$}\;
        }
    }
\end{algorithm}
