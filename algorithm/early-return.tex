\begin{algorithm}[H]
    \caption{Early-Exit Edges Detection}
    \label{algo:runtime-shape-check-detection}
    \footnotesize
    \SetKwFunction{RunOnFunction}{RunOnFunction}
    \SetKwFunction{MarkEarlyExitEdge}{MarkEarlyExitEdge}
    \SetKwFunction{TraceGuardingBranch}{TraceGuardingBranch}
    \SetKwFunction{GetPhiOperands}{GetPhiOperands}
    \SetKwFunction{PushBack}{PushBack}
    \SetKwFunction{Pop}{Pop}
    \SetKwFunction{IsPhi}{IsPhi}
    \SetKwFunction{IsCall}{IsCall}
    \SetKwFunction{IsConstant}{IsConstant}
    \SetKwFunction{FindBranchUser}{FindBranchUser}
    \SetKwFunction{GetFailSuccessor}{GetFailSuccessor}
    \SetKwFunction{ParentBlock}{ParentBlock}
    \SetKwProg{Fn}{Function}{:}{}
    
    \Fn{\RunOnFunction{$F,\;\mathcal{E},\;\mathcal{S}$}}{
        $B_{exit} \gets$ exit block of $F$\;
        $Inst_{ret} \gets$ return instruction in $B_{exit}$\;
        $v_{ret} \gets$ operand of $Inst_{ret}$\;
        
        $W \gets \varnothing$\;
        \uIf{\IsPhi{$v_{ret}$}}{ $W \gets \GetPhiOperands(v_{ret})$\; }
        \uElse{ $W.\PushBack(v_{ret}, B_{exit}, \text{null})$\; }
        
        \While{$W \neq \varnothing$}{
            $(v, B_{in}, B_{to}) \gets W.\Pop{}$\;
            
            \uIf{\IsPhi{$v$}}{
                $W.\PushBack(\GetPhiOperands(v))$\;
            }
            \uElseIf{\IsConstant{$v$}}{
                \If{$v \in \mathcal{E}$}{
                    \MarkEarlyExitEdge($B_{in} \to B_{to}$)\;
                    \TraceGuardingBranch($F, B_{in}, \mathcal{E}, \mathcal{S}$)\;
                }
            }
            \uElseIf{\IsCall{$v$}}{
                \RunOnFunction{callee of $v$, $\mathcal{E},\ \mathcal{S}$}\;
                
                $br \gets \FindBranchUser(v)$\;
                \If{$br \neq \text{null}$}{
                    $B_{fail} \gets \GetFailSuccessor(br, v, \mathcal{E}, \mathcal{S})$\;
                    \If{$B_{fail} \neq \text{null}$}{ 
                        \MarkEarlyExitEdge($\text{ParentBlock}(br) \to B_{fail}$)\; 
                    }
                }
            }
        }
    }
\end{algorithm}
